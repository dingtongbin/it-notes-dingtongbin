# 批量写与分区存储

表结构定好了，下一步是让它能扛住写入洪峰。这一节解决四个问题：怎么批量写才不被打穿、历史数据怎么低成本保留、原始数据怎么压缩、查数据时怎么避开全表扫。这套组合撑到日增千万点没有压力。

## 批量写管道（核心一）

写入路径的敌人是"逐条 INSERT"：每行一次网络往返 + 一次事务提交（redo/binlog fsync），云盘上典型只有 200~500 行/s。攒成 500 行一条多 VALUES 语句，fsync 成本摊到 500 行，典型 2 万~5 万行/s——两个数量级的差距。管道三要素：内存攒批（条数/时间双阈值）→ 多 VALUES 批量 INSERT → 失败指数退避重试。

```go
// store/writer.go
package store

import (
    "context"
    "database/sql"
    "log/slog"
    "strings"
    "sync"
    "sync/atomic"
    "time"
)

type Point struct {
    Series uint64
    Ts     time.Time
    Val    float64
}

// Writer 攒批写管道。调用方是流接收循环与探活结果回调——
// 它们都不能被落库阻塞，所以 Add 只碰内存，落库是 Run 的事
type Writer struct {
    db      *sql.DB
    mu      sync.Mutex
    buf     []Point
    maxBuf  int // 条数上限（有界）：OOM 的保险丝，超限丢老保新
    batch   int // 单条 INSERT 的行数
    dropped atomic.Uint64
}

func NewWriter(db *sql.DB, maxBuf, batch int) *Writer {
    return &Writer{db: db, maxBuf: maxBuf, batch: batch}
}

// Add 微秒级返回：只进内存缓冲。满则丢老保新——与 agent 缓冲同一策略，
// DB 故障时优先保住最新数据，丢弃必须计数
func (w *Writer) Add(pts []Point) {
    w.mu.Lock()
    w.buf = append(w.buf, pts...)
    if len(w.buf) > w.maxBuf {
        drop := len(w.buf) - w.maxBuf
        copy(w.buf, w.buf[drop:]) // 丢最旧：尾部 maxBuf 条前移
        w.buf = w.buf[:w.maxBuf]
        w.dropped.Add(uint64(drop))
        slog.Warn("写缓冲已满，丢老保新", "dropped", drop)
    }
    w.mu.Unlock()
}

// Run 后台落库循环：时间阈值 1s（攒不满也定期发），失败指数退避
func (w *Writer) Run(ctx context.Context) error {
    tick := time.NewTicker(time.Second)
    defer tick.Stop()
    backoff, fail := time.Duration(0), 0
    for {
        select {
        case <-ctx.Done():
            _ = w.flushOnce(context.Background()) // 优雅退出：送走残留；注意换新 ctx，原 ctx 已取消
            return nil
        case <-tick.C:
            if backoff > 0 { // 退避中：跳过本轮 tick，时间到再试
                backoff -= time.Second
                continue
            }
            if err := w.flushOnce(ctx); err != nil {
                fail++
                backoff = min(time.Duration(1<<fail)*time.Second, 30*time.Second) // 2s 4s 8s…封顶 30s
                slog.Error("批量写失败，退避重试", "err", err, "backoff", backoff)
            } else {
                fail = 0
            }
        }
    }
}

func (w *Writer) flushOnce(ctx context.Context) error {
    w.mu.Lock()
    if len(w.buf) == 0 {
        w.mu.Unlock()
        return nil
    }
    pts := w.buf
    w.buf = nil // 交出后置 nil，与后续 Add 不共享底层数组
    w.mu.Unlock()

    for start := 0; start < len(pts); start += w.batch {
        end := min(start+w.batch, len(pts))
        if err := w.insertChunk(ctx, pts[start:end]); err != nil {
            // 失败块塞回缓冲头部（退避后下次先写），已成功的块不回头
            w.mu.Lock()
            w.buf = append(pts[start:end], w.buf...)
            w.mu.Unlock()
            return err
        }
    }
    return nil
}

// insertChunk 多 VALUES 单条 INSERT：500 行一个事务、一次网络往返
func (w *Writer) insertChunk(ctx context.Context, pts []Point) error {
    var b strings.Builder
    b.WriteString("INSERT INTO metric_points (series_id, ts, val) VALUES ")
    args := make([]any, 0, len(pts)*3) // 预分配：热路径不做增量扩容
    for i, p := range pts {
        if i > 0 {
            b.WriteByte(',')
        }
        b.WriteString("(?,?,?)")
        args = append(args, p.Series, p.Ts.UTC(), p.Val) // 统一 UTC 落库
    }
    cctx, cancel := context.WithTimeout(ctx, 5*time.Second) // SQL 必须有超时
    defer cancel()
    _, err := w.db.ExecContext(cctx, b.String(), args...)
    return err
}
```

双阈值的含义：batch=500 是"攒满就发"（高峰吞吐），tick=1s 是"再少也发"（低谷延迟）。只有条数阈值的话，低峰期数据会躺在缓冲里好几分钟，最新值查询全是不新鲜的数据。

## 分区表：按天 RANGE，保留 = DROP

```sql
-- 分区 DDL：按天 RANGE；pmax 兜底防止越界写入报错
ALTER TABLE metric_points
PARTITION BY RANGE (TO_DAYS(ts)) (
    PARTITION p20260901 VALUES LESS THAN (TO_DAYS('2026-09-02')),
    PARTITION p20260902 VALUES LESS THAN (TO_DAYS('2026-09-03')),
    PARTITION pmax      VALUES LESS THAN MAXVALUE
);

-- 每天定时任务（提前建好明天的分区，用 REORGANIZE 拆 pmax）
ALTER TABLE metric_points REORGANIZE PARTITION pmax INTO (
    PARTITION p20260903 VALUES LESS THAN (TO_DAYS('2026-09-04')),
    PARTITION pmax VALUES LESS THAN MAXVALUE
);

-- 保留 30 天：整分区 DROP
ALTER TABLE metric_points DROP PARTITION p20260801;
```

查询自动裁剪验证：

```sql
EXPLAIN PARTITIONS
SELECT ts, val FROM metric_points
WHERE series_id = 42 AND ts >= '2026-09-01 00:00:00' AND ts < '2026-09-02 00:00:00';
-- partitions 列显示只扫 p20260901：31 个分区裁掉 30 个
```

保留策略为什么用 DROP PARTITION 而不是 DELETE：删一千万行是超大事务——undo 暴涨、行锁成片、binlog 巨大、主从延迟拉爆；DROP PARTITION 是 DDL 元数据操作，秒级完成、不产生大事务，差的不止百倍。注意分区键必须在主键里，所以主键是 (series_id, ts) 而不能只用 series_id——ts 恰好也在，天然满足。

## 降采样 rollup：10s 原始到 5min 聚合

原始 10s 采样保存一个月，趋势查询"最近 30 天 CPU"要扫 26 万点/序列——聚合表把量级砍掉 30 倍：

```sql
CREATE TABLE metric_points_5m (
    series_id BIGINT UNSIGNED NOT NULL,
    bucket    DATETIME(6) NOT NULL,  -- 窗口起点（对齐 5 分钟）
    min_v DOUBLE NOT NULL, max_v DOUBLE NOT NULL,
    avg_v DOUBLE NOT NULL, sum_v DOUBLE NOT NULL,
    cnt   INT NOT NULL,
    PRIMARY KEY (series_id, bucket)
) ENGINE=InnoDB;
```

```go
// store/rollup.go
package store

import (
    "context"
    "database/sql"
    "log/slog"
    "time"
)

// Rollupper 每分钟醒来一次，处理所有"已闭合"的 5 分钟窗口。
// 闭合判断：窗口终点 + 容忍期 <= now。容忍期（如 2 分钟）等迟到数据——
// agent 断网重连后补传的批次不能丢在窗口外
type Rollupper struct {
    db        *sql.DB
    last      time.Time // 上一个已聚合窗口的起点，从 job_state 表加载
    tolerance time.Duration
}

func (r *Rollupper) Run(ctx context.Context) error {
    tick := time.NewTicker(time.Minute)
    defer tick.Stop()
    for {
        select {
        case <-ctx.Done():
            return nil
        case <-tick.C:
            for {
                next := r.last.Add(5 * time.Minute)
                // 窗口未闭合（含容忍期）：下一轮再看
                if next.Add(5*time.Minute).Add(r.tolerance).After(time.Now()) {
                    break
                }
                if err := r.roll(ctx, next); err != nil {
                    slog.Error("rollup 失败，下轮重试", "bucket", next, "err", err)
                    break // 不推进 r.last：失败窗口下轮重做，闭合判断防重复聚合
                }
                r.last = next
            }
        }
    }
}

func (r *Rollupper) roll(ctx context.Context, bucket time.Time) error {
    const q = `INSERT INTO metric_points_5m (series_id, bucket, min_v, max_v, avg_v, sum_v, cnt)
               SELECT series_id, ?, MIN(val), MAX(val), AVG(val), SUM(val), COUNT(*)
               FROM metric_points
               WHERE ts >= ? AND ts < ?
               GROUP BY series_id
               ON DUPLICATE KEY UPDATE min_v=VALUES(min_v), max_v=VALUES(max_v),
                    avg_v=VALUES(avg_v), sum_v=VALUES(sum_v), cnt=VALUES(cnt)`
    // ON DUPLICATE 让重跑覆盖旧结果：幂等是"失败安全重试"的前提
    cctx, cancel := context.WithTimeout(ctx, 60*time.Second)
    defer cancel()
    _, err := r.db.ExecContext(cctx, q, bucket.UTC(), bucket.UTC(),
        bucket.Add(5*time.Minute).UTC())
    return err
}
```

窗口闭合判断是防重复聚合的关键：只有"窗口终点 + 容忍期"都过去了才聚合，配合 ON DUPLICATE 双保险——即使重跑也不会出双份。r.last 落在 job_state 表（name, last_done），重启续传。

## 查询接口

```sql
-- 区间序列（跨度小走原始表）
SELECT ts, val FROM metric_points
WHERE series_id = ? AND ts >= ? AND ts < ? ORDER BY ts;

-- 区间序列（跨度大走 rollup 表，曲线页按 step 自动选择）
SELECT bucket, avg_v FROM metric_points_5m
WHERE series_id = ? AND bucket >= ? AND bucket < ? ORDER BY bucket;

-- 最新值（MySQL 8 窗口函数，一次查 N 个序列的各自最新）
SELECT series_id, ts, val FROM (
    SELECT series_id, ts, val,
           ROW_NUMBER() OVER (PARTITION BY series_id ORDER BY ts DESC) AS rn
    FROM metric_points
    WHERE series_id IN (?, ?, ?) AND ts >= NOW() - INTERVAL 15 MINUTE
) t WHERE rn = 1;
```

p95 的等宽桶方案——时序库 histogram 思想的 MySQL 落地：精确分位要全量排序 O(n log n)，等宽桶把值域切成固定宽度的桶，只按桶计数：

```sql
-- RTT 的 p95：桶宽 10ms，bucket = FLOOR(val / 10) * 10
SELECT FLOOR(val / 10) * 10 AS bucket, COUNT(*) AS cnt
FROM metric_points
WHERE series_id = ? AND ts >= ? AND ts < ?
GROUP BY bucket ORDER BY bucket;
-- 应用层从低到高累加 cnt，累加值首次达到总数 95% 的桶上沿即近似 p95
```

精度换速度：桶宽越小越准、桶数越多越慢。RTT 这种有界指标用固定桶宽很合适。

## 何时换专业 TSDB

出现以下任一信号，就该把 DataSource 从 MySQL 换成 VictoriaMetrics 了：

写入持续超过 5k points/s（MySQL 批量写理论上 2~5 万/s，但要给查询和 rollup 留余量）。

series 数逼近百万（series 表查询与进程内缓存开始吃力）。

大盘查询（多序列长区间聚合）秒级超时成为常态，rollup 也救不动。

分区维护、rollup 链条的人力成本超过引入新组件的成本。

VictoriaMetrics 架构一瞥：写入先进内存 memtable，攒满刷成不可变的有序 part（LSM 思想），后台不断合并小 part 压空间；查询时先走倒排索引（指标名 + 标签 → series id 集合），再按时间块读列式数据。单二进制部署、资源占用低、原生说 PromQL——数据模型不变，gomon 只需替换 DataSource 实现。

## 常见坑

第一个坑是高基数标签打爆 series 数：tags 里混进 pid=1234、request_id 这类字段，进程一重启就是一个新 series。规则：tags 只允许低基数字段，高基数维度单独建表分析，不进时序主链路。

第二个坑是无界批量缓冲 OOM：DB 故障半小时，"先在内存里攒着"的缓冲默默吃掉几个 GB。必须有 maxBuf + 丢老保新 + 丢弃计数三件套，丢得明明白白比 OOM 强。

第三个坑是 DELETE 大事务清历史：DELETE FROM 删千万行 = 锁 + 巨型 binlog + 主从延迟。分区表的老数据只配 DROP PARTITION，DELETE 在时序场景几乎总是错的。

第四个坑是时间戳时区混乱：存储与传输统一 UTC，只在展示层转本地时区。DATETIME(6) 存的是 wall clock，跨 DB 会话迁移前先统一会话时区。

第五个坑是 GORM 逐条 Create 写时序：每条一个事务一次往返，吞吐直接掉回百级每秒。时序写入必须裸 database/sql 拼 VALUES 或用 CreateInBatches，批量是底线不是优化。

第六个坑是主键顺序写反：(ts, series_id) 做主键，同序列的点在 B+ 树上按时间全局散开，查询变随机 IO。必须 (series_id, ts)。
