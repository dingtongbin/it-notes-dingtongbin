# 指标上报与优雅退出

上一章讲了采集端怎么取数据，这一章解决数据怎么从 agent 可靠地到达 server：gRPC 双向流协议、有界缓冲、攒批双阈值 flush、持久化身份、被动模式应答、优雅退出。读完你会拥有一台不会丢数据、不会泄漏 goroutine、能自愈的 agent 主循环。

## 上报协议：proto 定义

```protobuf
// proto/gomon.proto
syntax = "proto3";
package gomon;
option go_package = "gomon/proto;gomonpb";

message Metric {
  string name = 1;              // 如 cpu.percent
  map<string, string> tags = 2; // 维度：如 mount="/data"——没有维度就没法告警
  double value = 3;
  int64 ts_ms = 4;
}

message MetricBatch {
  string agent_id = 1;          // 持久化唯一 ID，防重启变新资产
  string hostname = 2;
  int64 seq = 3;                // 单调递增：server 靠它检测传输缺口
  repeated Metric metrics = 4;
  map<string, string> agent_tags = 5; // os/arch/版本：每批都带，server 按 id upsert
}

message Ack { int64 seq = 1; }

service AgentReport {
  // 双向流：agent 上行推批次，server 下行回 Ack（预留流控指令下发）
  rpc Report(stream MetricBatch) returns (stream Ack);
}
```

心跳不需要单独协议：gRPC 流活着 + 每批携带元数据，就等于"注册 + 心跳"；server 把"流断开事件"视作失联信号，再用 Keepalive 探测半死连接。

## 有界缓冲：满了丢旧保新

监控数据宁可丢历史也不能丢当前。缓冲有上限，满了丢弃最旧的，并且丢弃必须计数并可观测。

```go
type Buffer struct {
    mu       sync.Mutex
    metrics  []Metric
    capacity int
    dropped  uint64 // 丢弃计数必须可观测，随批次上报
}

func (b *Buffer) Add(ms []Metric) (dropped int) {
    b.mu.Lock()
    defer b.mu.Unlock()
    b.metrics = append(b.metrics, ms...)
    if len(b.metrics) > b.capacity {
        dropped = len(b.metrics) - b.capacity
        copy(b.metrics, b.metrics[dropped:]) // 丢最旧：尾部 capacity 条前移
        b.metrics = b.metrics[:b.capacity]
        b.dropped += uint64(dropped)
    }
    return dropped
}

// Take 交出全部存量（清空），发送由调用方负责
func (b *Buffer) Take() []Metric {
    b.mu.Lock()
    defer b.mu.Unlock()
    out := b.metrics
    b.metrics = nil // 交出后置 nil，与后续 append 不共享底层数组
    return out
}

// PushFront 发送失败时塞回队头（断网期间数据不丢）；塞回超限仍丢最旧
func (b *Buffer) PushFront(ms []Metric) {
    b.mu.Lock()
    defer b.mu.Unlock()
    b.metrics = append(ms, b.metrics...)
    if len(b.metrics) > b.capacity {
        drop := len(b.metrics) - b.capacity
        copy(b.metrics, b.metrics[drop:])
        b.metrics = b.metrics[:b.capacity]
        b.dropped += uint64(drop)
    }
}
```

注意 Take 之后把切片置 nil 再返回：这样下一次 append 不会与已经交出去的底层数组共享内存，避免发送方和采集方向同一块底层数组写入引发数据竞争。

## 主循环：攒批与双阈值 flush

```go
type Agent struct {
    id         string // 持久化身份
    hostname   string
    client     gomonpb.AgentReportClient
    collectors []Collector
    buf        *Buffer
    interval   time.Duration
    seq        int64 // 仅主循环 goroutine 访问，无需加锁
}

func NewAgent(id, serverAddr string) (*Agent, error) {
    // Keepalive 探测半死连接（对端断电不发 FIN）：没有它流会"假活"几小时
    conn, err := grpc.NewClient(serverAddr,
        grpc.WithTransportCredentials(insecure.NewCredentials()),
        grpc.WithKeepaliveParams(keepalive.ClientParameters{
            Time: 30 * time.Second, Timeout: 10 * time.Second,
        }))
    if err != nil {
        return nil, err
    }
    host, _ := os.Hostname()
    return &Agent{
        id: id, hostname: host, client: gomonpb.NewAgentReportClient(conn),
        collectors: []Collector{&CPUCollector{}, &MemCollector{}, &DiskCollector{}},
        buf:        NewBuffer(10000), // 约 20 轮满负荷余量：断网 3 分钟不丢
        interval:   10 * time.Second,
    }, nil
}

// Run 外层重连：流断了退避重试，期间数据继续进有界缓冲
func (a *Agent) Run(ctx context.Context) error {
    for {
        if err := a.runOnce(ctx); err == nil || ctx.Err() != nil {
            return nil // 正常退出不算错误
        }
        slog.Error("流断开，5 秒后重连", "err", err)
        select {
        case <-time.After(5 * time.Second):
        case <-ctx.Done():
            return nil
        }
    }
}

func (a *Agent) runOnce(ctx context.Context) error {
    stream, err := a.client.Report(ctx) // gRPC 双向流
    if err != nil {
        return err
    }
    tick := time.NewTicker(a.interval)
    defer tick.Stop()
    const flushRounds = 8 // 时间阈值：8 轮（约 80s）攒一次批
    const maxBatch = 512  // 条数阈值：攒满即发——双阈值先到先触发
    round := 0
    for {
        select {
        case <-ctx.Done():
            a.flush(stream) // 残留数据送走再退
            stream.CloseSend()
            return ctx.Err()
        case <-tick.C:
            round++
            a.collectOnce(ctx)
            if round >= flushRounds || a.buf.Len() >= maxBatch {
                round = 0
                if err := a.flush(stream); err != nil {
                    return err // 流坏了：外层重连，数据已塞回缓冲
                }
            }
        }
    }
}

// collectOnce 一轮采集：统一 5s 超时——任何 collector 卡死
// （如 NFS stat 挂起）都不能拖垮下一轮 tick
func (a *Agent) collectOnce(ctx context.Context) {
    cctx, cancel := context.WithTimeout(ctx, 5*time.Second)
    defer cancel()
    for _, c := range a.collectors {
        if d := a.buf.Add(c.Collect(cctx)); d > 0 {
            slog.Warn("缓冲已满，丢旧保新", "collector", c.Name(), "dropped", d)
        }
    }
}

func (a *Agent) flush(stream gomonpb.AgentReport_ReportClient) error {
    ms := a.buf.Take()
    if len(ms) == 0 {
        return nil
    }
    a.seq++
    batch := &gomonpb.MetricBatch{
        AgentId: a.id, Hostname: a.hostname, Seq: a.seq,
        AgentTags: map[string]string{"os": runtime.GOOS, "arch": runtime.GOARCH},
        Metrics:   toPb(ms), // []Metric 转 proto，略
    }
    if err := stream.Send(batch); err != nil {
        a.buf.PushFront(ms) // 失败塞回队头，重连后下一轮带走
        return err
    }
    return nil
}
```

双阈值的含义：flushRounds 是"再少也发"（保证低谷延迟），maxBatch 是"攒满就发"（控制单批体积与内存）。只靠时间阈值，高峰期的批次会越来越大；只靠条数阈值，低峰期数据会躺在缓冲里拖延数分钟不新鲜。

## agent_id：持久化的身份

```go
// loadOrCreateID 生成一次、落盘复用：否则每次重启都被 server 当成新资产，
// 历史曲线断成两截、告警状态全部重置
func loadOrCreateID(path string) (string, error) {
    if b, err := os.ReadFile(path); err == nil {
        if id := strings.TrimSpace(string(b)); id != "" {
            return id, nil
        }
    }
    b := make([]byte, 16)
    if _, err := rand.Read(b); err != nil { // crypto/rand 系统级随机源
        return "", err
    }
    id := hex.EncodeToString(b)
    return id, os.WriteFile(path, []byte(id), 0o600)
}
```

更稳的身份是 Linux 的 /etc/machine-id（重装系统才会变）：有它就读它，没有再随机生成。

## server 端：流接收骨架

```go
// Report 是 Zabbix trapper 的对应物：接流、检测缺口、入库、回 Ack
func (s *Server) Report(stream gomonpb.AgentReport_ReportServer) error {
    for {
        batch, err := stream.Recv()
        if err != nil {
            return err // 流断开：agent 退出或网络故障；失联事件由此触发
        }
        // 批次 seq 连续性检查：缺口 = 传输丢批
        if last, ok := s.lastSeq.Load(batch.AgentId); ok { // lastSeq 为 sync.Map
            if expect := last.(int64) + 1; batch.Seq != expect {
                slog.Warn("批次缺口", "agent", batch.AgentId,
                    "expect", expect, "got", batch.Seq)
            }
        }
        s.lastSeq.Store(batch.AgentId, batch.Seq)
        // 入库交给存储层：失败不 return——DB 抖动不该断流，
        // 失败批由存储层落盘重放
        if err := s.store.Write(stream.Context(), batch); err != nil {
            slog.Error("入库失败", "agent", batch.AgentId, "err", err)
        }
        if err := stream.Send(&gomonpb.Ack{Seq: batch.Seq}); err != nil {
            return err
        }
    }
}
```

## 被动模式：TCP + 长度前缀 JSON

场景：NAT 后无法出站的 agent，或 server 想立即要一份数据不等下一轮。协议复用 TCP 网络编程的长度前缀编解码：

```
请求（server → agent）：4 字节大端长度 + {"cmd":"collect"}
应答（agent → server）：4 字节大端长度 + {"agent_id":"...","metrics":[...]}
```

```go
func (a *Agent) ServePassive(ctx context.Context, addr string) error {
    var lc net.ListenConfig // ListenConfig 原生支持 ctx：取消即关闭，退出路径干净
    ln, err := lc.Listen(ctx, "tcp", addr)
    if err != nil {
        return err
    }
    go func() {
        <-ctx.Done()
        ln.Close() // ctx 取消时关闭监听，Accept 报错，循环自然退出
    }()
    for {
        conn, err := ln.Accept()
        if err != nil {
            if ctx.Err() != nil {
                return nil // 主动关闭：正常退出
            }
            return err
        }
        go a.handlePassive(ctx, conn)
    }
}

func (a *Agent) handlePassive(ctx context.Context, conn net.Conn) {
    defer conn.Close()
    conn.SetDeadline(time.Now().Add(10 * time.Second)) // 僵死连接必须有超时
    var head [4]byte
    if _, err := io.ReadFull(conn, head[:]); err != nil {
        return
    }
    if n := binary.BigEndian.Uint32(head[:]); n > 1<<20 {
        return // 长度上限：防恶意超长包打爆内存
    } else if _, err := io.ReadFull(conn, make([]byte, n)); err != nil {
        return // 请求体不必解析：被动模式只有一个命令 collect
    }
    // 现场采集一轮：被动模式要的是"此刻"状态，不是缓冲里的旧数据
    cctx, cancel := context.WithTimeout(ctx, 5*time.Second)
    defer cancel()
    ms := make([]Metric, 0, 64)
    for _, c := range a.collectors {
        ms = append(ms, c.Collect(cctx)...)
    }
    out, _ := json.Marshal(struct {
        AgentID string   `json:"agent_id"`
        Metrics []Metric `json:"metrics"`
    }{a.id, ms})
    binary.BigEndian.PutUint32(head[:], uint32(len(out)))
    conn.Write(head[:])
    conn.Write(out)
}
```

连接上有两处有界约束：长度字段拒绝了恶意超长包；SetDeadline 10 秒拒绝僵死连接占着 goroutine 不放。

## 优雅退出与组装

退出路径：SIGTERM 自动取消 ctx，select 捕获后 flush 残留数据、CloseSend、返回。残留数据不送走，每次重启都会丢最后一轮，所以退出路径必须兜住。

```go
func main() {
    // Ctrl+C / SIGTERM 自动取消 ctx：采集超时、被动监听、gRPC 流一起收到信号
    ctx, stop := signal.NotifyContext(context.Background(), syscall.SIGINT, syscall.SIGTERM)
    defer stop()
    id, err := loadOrCreateID("gomon-agent.id")
    if err != nil {
        log.Fatal(err)
    }
    agent, err := NewAgent(id, "127.0.0.1:9100")
    if err != nil {
        log.Fatal(err)
    }
    // 生产上 Run 与 ServePassive 各占一个 goroutine，WaitGroup 等齐后退出
    if err := agent.Run(ctx); err != nil {
        slog.Error("agent 退出", "err", err)
    }
}
```

ListenConfig.Listen 接受 ctx、signal.NotifyContext 自动取消 ctx，两者配合让"关监听、断流、flush 残留"全部由一条取消链路驱动，退出路径没有遗漏。

## 常见坑

第一个坑是时间戳时区与时区混淆：agent 打本地时间戳、server 存 DATETIME、前端按又一时区解析，曲线会错 8 小时。铁律：存储与传输统一 UTC，只在展示层转本地时区。

第二个坑是 gRPC 双向流的状态语义：Send 成功只代表字节进了内核发送缓冲，不代表对端收到。真正的进度确认靠对端的持久化 seq，断线重连后要对齐。

第三个坑是提交的 seq 不回退：flush 之前先 seq++，失败塞回缓冲，重连后重发的是同一个 seq，server 靠幂等去重。绝不能重连后又生成新 seq。

第四个坑是某台机器时钟漂移污染全链路：所有 agent 用自己的时钟打戳，一台 NTP 失步的机器会把"未来"时间戳混进序列，破坏窗口计算。对外提供心跳同步，或对异常时间戳做钳制。

第五个坑是信号处理顺序：Stop 必须在取消之后调用，否则信号到达时 ctx 已经提前取消，优雅退出的 flush 分支根本不会执行。
